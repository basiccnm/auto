# 네이버쇼핑 인기순위 위젯 (Blogspot 사이드바 전용)

## 설치 방법
1. Blogger 대시보드 → 레이아웃 → 사이드바 영역에서 **"가젯 추가" → "HTML/자바스크립트"**
2. 제목은 비워두고, 아래 코드 전체를 내용에 붙여넣기 → 저장
3. (v4 업데이트, 2026-07-09) 이미 설치했다면 아래 코드로 전체 교체 — 카드 테두리를
   사이드바 가젯 v4가 담당하게 바뀌어서 이중 테두리 방지용 수정이 있습니다

## 동작 방식
- 하루 1회 갱신되는 정적 JSON(GitHub)을 읽어서 표시 (실시간 아님 — "전날 검색 인기순" 표기)
- 현재 보고 있는 글의 라벨(유산균/오메가3/루테인/갱년기/다이어트/남성건강/단백질)에 맞는 순위를 자동 표시
- 라벨이 없는 페이지(홈 등)는 유산균 순위를 기본 표시
- 항목 클릭 시 아무 데도 연결되지 않음 (순수 정보성 — 의도된 동작)

## 데이터 갱신
- 관리자 페이지(localhost:8899) 대시보드의 "위젯 순위 갱신" 버튼, 또는 `python naver_widget.py`
- 하루 1회 실행 권장

```html
<div id="bp-shop-rank" style="font-family:'Noto Sans KR','Malgun Gothic',sans-serif;"></div>
<script>
(function () {
  var JSON_URL = 'https://raw.githubusercontent.com/bangpharmacy-images/bangpharmacy-images/main/images/img_ranking.json';
  var DEFAULT_KEY = '유산균';

  function pageLabels() {
    // 현재 페이지의 라벨 집합을 모은다 (본문 영역 .widget.Blog 안의 라벨만 —
    // 사이드바 라벨 목록까지 스캔하면 가나다순 첫 라벨로 오매칭됨)
    var labels = [];
    if (location.pathname.indexOf('/search/label/') !== -1) {
      var u = decodeURIComponent(location.pathname.split('/search/label/')[1] || '').split('?')[0];
      if (u) labels.push(u);
    }
    if (/\/20\d\d\//.test(location.pathname)) {
      var scope = document.querySelector('.widget.Blog') || document;
      var links = scope.querySelectorAll('a[href*="/search/label/"]');
      for (var i = 0; i < links.length; i++) {
        var t = decodeURIComponent(links[i].getAttribute('href').split('/search/label/')[1] || '').split('?')[0];
        if (t && labels.indexOf(t) === -1) labels.push(t);
      }
    }
    return labels;
  }

  function pickEntry(entries, labels) {
    // 글의 라벨이 entry.match를 전부 포함하는 항목 중, match 개수가 가장 많은(구체적인) 것
    var best = null;
    for (var i = 0; i < entries.length; i++) {
      var e = entries[i];
      var ok = e.match.every(function (m) { return labels.indexOf(m) !== -1; });
      if (ok && (!best || e.match.length > best.match.length)) best = e;
    }
    if (!best) {
      for (var j = 0; j < entries.length; j++) {
        if (entries[j].name === DEFAULT_KEY) { best = entries[j]; break; }
      }
    }
    return best || entries[0];
  }

  fetch(JSON_URL).then(function (r) { return r.json(); }).then(function (data) {
    var entries = data.entries || [];
    if (!entries.length) return;
    var entry = pickEntry(entries, pageLabels());
    var key = entry.name;
    var items = entry.items || [];
    if (!items.length) return;

    // 카드 테두리는 사이드바 가젯 v4가 위젯 차원에서 입히므로 여기선 내용만 (이중 테두리 방지)
    var html = '<div>'
      + '<div style="font-weight:bold;font-size:15px;color:#25a186;margin-bottom:2px;">🛒 ' + key + ' 인기 TOP' + items.length + '</div>'
      + '<div style="font-size:11px;color:#999;margin-bottom:8px;">네이버쇼핑 ' + (data.label || '전날 검색 인기순') + ' · ' + (data.updated || '') + ' 기준</div>'
      + '<ol style="margin:0;padding-left:0;list-style:none;">';
    for (var i = 0; i < items.length; i++) {
      var it = items[i];
      var price = it.lprice ? it.lprice.toLocaleString('ko-KR') + '원~' : '';
      html += '<li style="display:flex;gap:8px;align-items:baseline;padding:6px 0;border-top:1px solid #f2f2f2;font-size:13px;line-height:1.4;">'
        + '<span style="flex-shrink:0;font-weight:bold;color:' + (i < 3 ? '#e60000' : '#999') + ';">' + (i + 1) + '</span>'
        + '<span style="color:#333;">' + it.title + '<br><span style="font-size:12px;color:#888;">' + price + '</span></span></li>';
    }
    html += '</ol><div style="font-size:10px;color:#bbb;margin-top:6px;">※ 정보 제공 목적이며 구매 링크가 아닙니다</div></div>';
    document.getElementById('bp-shop-rank').innerHTML = html;
  }).catch(function () {});
})();
</script>
```
