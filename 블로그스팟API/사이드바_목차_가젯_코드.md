## v4 변경사항 (2026-07-09)
- 모든 사이드바 위젯(목차/인기게시물/쇼핑위젯)을 쇼핑위젯과 같은 **카드 스타일**로 통일
  (흰 배경 + 민트 테두리 + 둥근 모서리)
- **스마트폰 사이드바 잘림 수정**: 드로어 폭이 390px 고정이라 375px 화면에서 잘리던 것을
  화면의 86%로 자동 축소되게 변경

## 적용 방법
1. **기존 목차 가젯** 내용을 아래 코드로 전체 교체 → 저장
2. 예전 스타일 가젯(배경/인기게시물용 별도 HTML 가젯)이 남아 있으면 삭제 (충돌 방지)

<script>
(function(){
  var box=document.getElementById('bp-toc');
  var here=location.pathname.replace(/\/$/,'');
  fetch('/p/blog-page_321.html')
    .then(function(r){return r.text();})
    .then(function(html){
      var doc=new DOMParser().parseFromString(html,'text/html');
      var body=doc.querySelector('.post-body');
      if(!body){box.textContent='목차를 불러올 수 없습니다';return;}
      var root=body.querySelector('div')||body;
      var nodes=root.children,sections=[],cur=null,secNo=0;
      for(var i=0;i<nodes.length;i++){
        var n=nodes[i];
        if(n.tagName==='H2'){
          secNo++;
          if(cur)sections.push(cur);
          cur={title:secNo+'. '+n.textContent.trim().replace(/\s*시리즈\s*$/,''),links:[]};
        }else if(cur&&n.querySelectorAll){
          var as=n.querySelectorAll('a');
          for(var j=0;j<as.length;j++){
            var a=as[j],d=0,p=a.parentElement;
            while(p&&p!==n){if(p.tagName==='UL')d++;p=p.parentElement;}
            var href=a.getAttribute('href')||'';
            var path='';
            try{path=new URL(href,location.origin).pathname.replace(/\/$/,'');}catch(e){}
            cur.links.push({t:a.textContent.trim(),h:href,d:d,on:(path&&path===here)});
          }
        }
      }
      if(cur)sections.push(cur);
      if(!sections.length){box.textContent='목차가 비어있습니다';return;}
      var out='<a class="bp-home" href="/">홈</a>';
      out+=sections.map(function(s){
        var open=s.links.some(function(l){return l.on;});
        var lis=s.links.map(function(l){
          var ind=(l.d>1)?' style="margin-left:'+((l.d-1)*10)+'px;"':'';
          var cls=l.on?' class="bp-active"':'';
          return '<li'+ind+'><a'+cls+' href="'+l.h+'">'+l.t+'</a></li>';
        }).join('');
        return '<details'+(open?' open':'')+'><summary>'+s.title+'</summary><ul>'+lis+'</ul></details>';
      }).join('');
      box.innerHTML=out;
    })
    .catch(function(){box.textContent='목차 로딩 실패 (새로고침 해주세요)';});
})();
</script>
